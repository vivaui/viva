package {{PACKAGE}}

import (
	"github.com/vivaui/viva/gfx"
	"github.com/vivaui/viva/style"
)

// ToggleVariant selects the neutral or bordered treatment.
type ToggleVariant uint8

const (
	ToggleDefault ToggleVariant = iota
	ToggleOutline
)

// ToggleSize selects compact relative control metrics.
type ToggleSize uint8

const (
	ToggleSizeDefault ToggleSize = iota
	ToggleSizeSmall
	ToggleSizeLarge
)

// ToggleRecipe resolves the look of a toggle of one variant in one state: the
// fill, the label, the line, the focus ring and the shadow. Checked is the
// active state; hover, press, focus and invalid are read only while the toggle
// is not disabled, and Disabled dims the fill, the label and the line by half.
func ToggleRecipe(theme gfx.Theme, variant ToggleVariant, state gfx.State) style.ButtonLook {
	radius := theme.Metrics.RadiusControl
	fill := theme.Background
	fill.Alpha = 0
	foreground := theme.Foreground
	border := theme.Input
	bordered := variant == ToggleOutline
	if state.Checked {
		fill = theme.Muted
	}
	if state.Hovered && !state.Disabled {
		fill = theme.Muted
		foreground = theme.Foreground
	}
	if state.Pressed && !state.Disabled {
		fill = blend(fill, theme.Foreground, 0.05)
	}
	look := style.ButtonLook{
		Background: fill, Foreground: foreground, Border: border,
		Filled: true, Bordered: bordered, Radius: radius,
	}
	if variant == ToggleOutline {
		look.Shadow = gfx.ShadowXS
	}
	// The component draws an invalid ring even on a disabled toggle; focus is
	// the keyboard's, so it stops when the toggle is unavailable.
	if state.Invalid || state.FocusVisible && !state.Disabled {
		ring, edge := scaledAlpha(theme.Ring, 0.50), theme.Ring
		if state.Invalid {
			ring, edge = invalidRing(&theme), invalidBorder(&theme)
		}
		look.Ring, look.RingWidth, look.RingRadius = ring, 3, radius+3
		look.Border, look.Bordered = edge, true
	}
	if state.Disabled {
		look.Background.Alpha *= 0.50
		look.Foreground.Alpha *= 0.50
		look.Border.Alpha *= 0.50
	}
	return look
}

// Toggle paints one two-state compact action. The application owns active
// state and responds to the stable Canvas hit ID.
type Toggle struct {
	canvas   *gfx.Canvas
	theme    gfx.Theme
	id       gfx.ID
	bounds   gfx.Rect
	active   bool
	size     ToggleSize
	invalid  bool
	disabled bool
}

// Small selects the compact size variant.
func (toggle Toggle) Small() Toggle {
	toggle.size = ToggleSizeSmall
	return toggle
}

// Large selects the large size variant.
func (toggle Toggle) Large() Toggle {
	toggle.size = ToggleSizeLarge
	return toggle
}

// Invalid applies the destructive validation treatment.
func (toggle Toggle) Invalid() Toggle {
	toggle.invalid = true
	return toggle
}

// BeginToggle starts a toggle declaration.
func BeginToggle(canvas *gfx.Canvas, theme gfx.Theme, id gfx.ID, bounds gfx.Rect, active bool) Toggle {
	return Toggle{canvas: canvas, theme: theme, id: id, bounds: bounds, active: active}
}

// Disabled renders and routes the toggle as inert.
func (toggle Toggle) Disabled() Toggle {
	toggle.disabled = true
	return toggle
}

// SizeToFit widens the toggle to its label plus Vega's px-2.5, never narrower
// than it is tall (min-w).
func (toggle Toggle) SizeToFit(label string) Toggle {
	height, padding, _ := style.ToggleGeometry(toggle.theme.Metrics, toggleStyleSize(toggle.size))
	toggle.bounds.Height = height
	toggle.bounds.Width = toggleFittedWidth(toggle.canvas, toggle.theme, label, height, padding)
	return toggle
}

// BoundsWidth exposes the fitted width for callers laying out neighbors.
func (toggle Toggle) BoundsWidth() float32 { return toggle.bounds.Width }

// toggleFittedWidth is a toggle's width around label: padding on both sides, at
// least as wide as it is tall.
func toggleFittedWidth(canvas *gfx.Canvas, theme gfx.Theme, label string, height, padding float32) float32 {
	return max(canvas.TextWidthWeight(label, theme.Metrics.FontSizeM, gfx.WeightMedium)+padding*2, height)
}

// ToggleResult is what painting a toggle reports back. Clicked is true in the
// event pass of a click on it (see gfx.Canvas.Clicked), so the code that declares
// a toggle is the code that flips it:
//
//	if components.BeginToggle(canvas, theme, bold, bounds, m.bold).Default("B").Clicked() {
//		m.bold = !m.bold
//	}
type ToggleResult struct {
	canvas *gfx.Canvas
	id     gfx.ID
}

// Clicked reports whether this paint is delivering a click on the toggle.
func (result ToggleResult) Clicked() bool {
	return result.canvas != nil && result.canvas.Clicked(result.id)
}

// Default paints the borderless treatment.
func (toggle Toggle) Default(label string) ToggleResult { return toggle.paint(label, ToggleDefault) }

// Outline paints the bordered treatment.
func (toggle Toggle) Outline(label string) ToggleResult { return toggle.paint(label, ToggleOutline) }

func (toggle Toggle) paint(label string, variant ToggleVariant) ToggleResult {
	result := ToggleResult{canvas: toggle.canvas, id: toggle.id}
	if toggle.canvas == nil {
		return result
	}
	if toggle.bounds.Height <= 0 {
		toggle.bounds.Height = toggle.controlHeight()
	}
	if toggle.bounds.Width <= 0 {
		toggle.bounds.Width = toggle.bounds.Height
	}
	state := gfx.State{Checked: toggle.active, Invalid: toggle.invalid, Disabled: toggle.disabled}
	if !toggle.disabled && !toggle.id.IsZero() {
		state.Hovered = toggle.canvas.Hovered() == toggle.id
		state.Pressed = toggle.canvas.Pressed() == toggle.id
		state.FocusVisible = toggle.canvas.FocusVisible() == toggle.id
	}
	look := ToggleRecipe(toggle.theme, variant, state)
	if look.Shadow != gfx.ShadowNone {
		toggle.canvas.Elevate(toggle.bounds, look.Radius, look.Shadow) // shadow-xs
	}
	if look.RingWidth > 0 {
		toggle.canvas.Rounded(expand(toggle.bounds, look.RingWidth), look.RingRadius, look.Ring)
	}
	if look.Bordered && look.Filled {
		paintBordered(toggle.canvas, toggle.theme, toggle.bounds, look.Radius, look.Border, look.Background)
	} else if look.Filled {
		toggle.canvas.Rounded(toggle.bounds, look.Radius, look.Background)
	}
	toggle.canvas.LabelCenter(label, toggle.bounds, gfx.TextStyle{
		Size: toggle.theme.Metrics.FontSizeM, Weight: gfx.WeightMedium, Color: look.Foreground,
	})
	if !toggle.id.IsZero() {
		toggle.canvas.HitWithOptions(toggle.id, toggle.bounds, label, gfx.HitOptions{
			Role: gfx.RoleButton, Disabled: toggle.disabled, Checked: toggle.active,
			Selected: toggle.active, Invalid: toggle.invalid,
		})
	}
	return result
}

func (toggle Toggle) controlHeight() float32 {
	return toggleControlHeight(toggle.theme, toggle.size)
}

func toggleControlHeight(theme gfx.Theme, size ToggleSize) float32 {
	height, _, _ := style.ToggleGeometry(theme.Metrics, toggleStyleSize(size))
	return height
}

// toggleStyleSize is the element size a component toggle size maps to; both
// the node measurement and the component's SizeToFit take their metrics from
// style.ToggleGeometry.
func toggleStyleSize(size ToggleSize) style.Size {
	switch size {
	case ToggleSizeSmall:
		return style.SizeSmall
	case ToggleSizeLarge:
		return style.SizeLarge
	}
	return style.SizeDefault
}
