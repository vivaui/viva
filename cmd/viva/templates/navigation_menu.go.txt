// NavigationMenuOptions describes caller-owned disclosure and popup geometry.
type NavigationMenuOptions struct {
	Open          bool
	DirectContent bool
	Content       gfx.Rect
	Layer         uint16
	Scope         gfx.ID
	DismissID     gfx.ID
	// Viewport, with DismissID, makes an open navigation menu dismiss on a press
	// outside it and passes that press on, as a non-modal navigation menu does.
	Viewport gfx.Rect
}

// NavigationMenuTrigger describes one stable top-level disclosure.
type NavigationMenuTrigger struct {
	ID       gfx.ID
	Label    string
	Hint     string
	Open     bool
	Active   bool
	Disabled bool
	Width    float32
}

// NavigationMenuLink describes a top-level or popup navigation destination.
type NavigationMenuLink struct {
	ID          gfx.ID
	Label       string
	Description string
	Value       string
	Hint        string
	Active      bool
	Disabled    bool
	Width       float32
}

// NavigationMenu paints horizontal triggers and one caller-selected popup.
type NavigationMenu struct {
	canvas  *gfx.Canvas
	theme   gfx.Theme
	bounds  gfx.Rect
	options NavigationMenuOptions
	x       float32
	active  gfx.Rect
}

// NavigationMenuContent paints the active popup contents.
type NavigationMenuContent struct {
	canvas  *gfx.Canvas
	theme   gfx.Theme
	bounds  gfx.Rect
	options NavigationMenuOptions
	y       float32
	visible bool
}

// BeginNavigationMenu begins one horizontal navigation declaration.
func BeginNavigationMenu(canvas *gfx.Canvas, theme gfx.Theme, bounds gfx.Rect, options NavigationMenuOptions) NavigationMenu {
	menu := NavigationMenu{canvas: canvas, theme: theme, bounds: bounds, options: options, x: bounds.X}
	if canvas != nil && options.Open && !options.DismissID.IsZero() {
		canvas.SetEscapeTarget(options.DismissID)
		if options.Viewport.Width > 0 && options.Viewport.Height > 0 {
			canvas.HitWithOptions(options.DismissID, options.Viewport, "Dismiss navigation", gfx.HitOptions{
				Role: gfx.RoleButton, SkipFocus: true, Backdrop: true, PassPrimary: true, PassSecondary: true,
			})
		}
	}
	return menu
}

// AddTrigger paints one official 36px navigation disclosure.
func (menu *NavigationMenu) AddTrigger(trigger NavigationMenuTrigger) gfx.Rect {
	if menu.canvas == nil || menu.bounds.Width <= 0 {
		return gfx.Rect{}
	}
	height := float32(36)
	width := trigger.Width
	if width <= 0 {
		width = menu.canvas.TextWidthWeight(trigger.Label, menu.theme.Metrics.FontSizeM, gfx.WeightMedium) + 48
	}
	if menu.x+width > menu.bounds.X+menu.bounds.Width {
		return gfx.Rect{}
	}
	bounds := gfx.Rect{X: menu.x, Y: menu.bounds.Y, Width: width, Height: height}
	menu.x += width // the list is gap-0
	if trigger.Open {
		menu.active = bounds
	}
	// hover and focus are bg-muted; an open trigger that is not hovered is bg-muted/50.
	interacted := !trigger.Disabled && !trigger.ID.IsZero() && (menu.canvas.Hovered() == trigger.ID || menu.canvas.FocusVisible() == trigger.ID)
	if interacted || trigger.Open || trigger.Active {
		color := menu.theme.Muted
		if !interacted {
			color.Alpha *= 0.50
		}
		menu.canvas.Rounded(bounds, menu.theme.Metrics.RadiusControl, color)
	}
	foreground := menu.theme.Foreground
	icon := menu.theme.MutedForeground
	if trigger.Disabled {
		foreground.Alpha *= 0.50
		icon.Alpha *= 0.50
	}
	menu.canvas.LabelCenter(trigger.Label, gfx.Rect{X: bounds.X + 12, Y: bounds.Y, Width: navigationMenuMax(0, bounds.Width-32), Height: bounds.Height}, gfx.TextStyle{Size: menu.theme.Metrics.FontSizeM, Weight: gfx.WeightMedium, Color: foreground})
	chevron := &chevronDownIcon
	if trigger.Open {
		chevron = &chevronUpIcon // rotate-180 while open
	}
	paintChevronSized(menu.canvas, chevron, gfx.Rect{X: bounds.X + bounds.Width - 20, Y: bounds.Y, Width: 12, Height: bounds.Height}, 12, icon)
	if !trigger.ID.IsZero() {
		menu.canvas.HitWithOptions(trigger.ID, bounds, trigger.Label, gfx.HitOptions{Role: gfx.RoleButton, Hint: trigger.Hint, Disabled: trigger.Disabled, Selected: trigger.Active, Expanded: trigger.Open, Expandable: true})
	}
	return bounds
}

// NavigationMenuKeyDown moves focus between top-level triggers: Left/Right with
// wrapping (reversed in RTL), Home/End to the ends, Down/Enter/Space opens the
// focused trigger, and Up/Escape closes the open one and returns to its trigger.
// It returns the trigger to focus and the trigger whose panel is open (zero for
// none), and whether it owned the key.
func NavigationMenuKeyDown(current gfx.ID, triggers []NavigationMenuTrigger, open gfx.ID, direction Direction, key gfx.Key) (gfx.ID, gfx.ID, bool) {
	enabled := func(index int) bool {
		trigger := triggers[index]
		return !trigger.ID.IsZero() && !trigger.Disabled
	}
	step := 0
	switch key {
	case gfx.KeyLeft:
		step = -1
		if direction.IsRTL() {
			step = 1
		}
	case gfx.KeyRight:
		step = 1
		if direction.IsRTL() {
			step = -1
		}
	case gfx.KeyHome, gfx.KeyEnd:
		index := -1
		if key == gfx.KeyHome {
			for at := 0; at < len(triggers); at++ {
				if enabled(at) {
					index = at
					break
				}
			}
		} else {
			for at := len(triggers) - 1; at >= 0; at-- {
				if enabled(at) {
					index = at
					break
				}
			}
		}
		if index < 0 {
			return current, open, false
		}
		next := triggers[index].ID
		if !open.IsZero() {
			open = next
		}
		return next, open, true
	case gfx.KeyDown, gfx.KeyEnter, gfx.KeySpace:
		if current.IsZero() {
			return current, open, false
		}
		return current, current, true
	case gfx.KeyUp, gfx.KeyEscape:
		if open.IsZero() {
			return current, open, false
		}
		return open, gfx.ID{}, true
	default:
		return current, open, false
	}
	index := 0
	for at := range triggers {
		if triggers[at].ID == current {
			index = at
			break
		}
	}
	for offset := 1; offset <= len(triggers); offset++ {
		at := (index + step*offset) % len(triggers)
		if at < 0 {
			at += len(triggers)
		}
		if !enabled(at) {
			continue
		}
		next := triggers[at].ID
		if !open.IsZero() {
			open = next
		}
		return next, open, true
	}
	return current, open, false
}

// AddLink paints a top-level navigation link using the same trigger geometry.
func (menu *NavigationMenu) AddLink(link NavigationMenuLink) gfx.Rect {
	if menu.canvas == nil || menu.bounds.Width <= 0 {
		return gfx.Rect{}
	}
	width := link.Width
	if width <= 0 {
		width = menu.canvas.TextWidthWeight(link.Label, menu.theme.Metrics.FontSizeM, gfx.WeightMedium) + 32
	}
	if menu.x+width > menu.bounds.X+menu.bounds.Width {
		return gfx.Rect{}
	}
	bounds := gfx.Rect{X: menu.x, Y: menu.bounds.Y, Width: width, Height: 36}
	menu.x += width
	interacted := !link.Disabled && !link.ID.IsZero() && (menu.canvas.Hovered() == link.ID || menu.canvas.FocusVisible() == link.ID)
	if interacted || link.Active {
		color := menu.theme.Muted
		if !interacted {
			color.Alpha *= 0.50 // data-[active=true]:bg-muted/50
		}
		menu.canvas.Rounded(bounds, menu.theme.Metrics.RadiusControl, color)
	}
	foreground := menu.theme.Foreground
	if link.Disabled {
		foreground.Alpha *= 0.50
	}
	menu.canvas.LabelCenter(link.Label, bounds, gfx.TextStyle{Size: menu.theme.Metrics.FontSizeM, Weight: gfx.WeightMedium, Color: foreground})
	if !link.ID.IsZero() {
		menu.canvas.HitWithOptions(link.ID, bounds, link.Label, gfx.HitOptions{Role: gfx.RoleLink, Value: link.Value, Hint: link.Hint, Disabled: link.Disabled, Selected: link.Active})
	}
	return bounds
}

// BeginContent paints the active viewport/direct popup and returns its row API.
func (menu NavigationMenu) BeginContent() NavigationMenuContent {
	content := NavigationMenuContent{canvas: menu.canvas, theme: menu.theme, options: menu.options}
	if menu.canvas == nil || !menu.options.Open {
		return content
	}
	bounds := menu.options.Content
	if bounds.Width <= 0 {
		bounds.Width = 420
	}
	if bounds.Height <= 0 {
		bounds.Height = 220
	}
	if bounds.X == 0 && bounds.Y == 0 {
		bounds.X = menu.bounds.X
		bounds.Y = menu.bounds.Y + 42
	}
	content.bounds = bounds
	content.y = bounds.Y + 8
	content.visible = true
	content.push()
	paintPopupSurface(menu.canvas, &menu.theme, bounds, gfx.ShadowSM) // shadow (= shadow-sm)
	if !menu.options.DirectContent && menu.active.Width > 0 {
		indicator := gfx.Rect{X: menu.active.X + menu.active.Width/2 - 4, Y: bounds.Y - 4, Width: 8, Height: 4}
		menu.canvas.RoundedBorder(indicator, 2, 1, menu.theme.Border, menu.theme.Popover)
	}
	content.pop()
	return content
}

// Bounds returns the popup viewport geometry.
func (content NavigationMenuContent) Bounds() gfx.Rect { return content.bounds }

// AddLink paints a stacked title/description destination.
func (content *NavigationMenuContent) AddLink(link NavigationMenuLink) gfx.Rect {
	if !content.visible {
		return gfx.Rect{}
	}
	height := float32(56)
	if link.Description == "" {
		height = 36
	}
	if content.y+height > content.bounds.Y+content.bounds.Height-8 {
		return gfx.Rect{}
	}
	bounds := gfx.Rect{X: content.bounds.X + 8, Y: content.y, Width: navigationMenuMax(0, content.bounds.Width-16), Height: height}
	content.y += height + 4
	content.push()
	interacted := !link.Disabled && !link.ID.IsZero() && (content.canvas.Hovered() == link.ID || content.canvas.FocusVisible() == link.ID)
	if interacted || link.Active {
		color := content.theme.Muted
		if !interacted {
			color.Alpha *= 0.50
		}
		content.canvas.Rounded(bounds, content.theme.Metrics.RadiusItem(), color)
	}
	foreground := content.theme.PopoverForeground
	description := content.theme.MutedForeground
	if link.Disabled {
		foreground.Alpha *= 0.50
		description.Alpha *= 0.50
	}
	if link.Description == "" {
		content.canvas.Label(link.Label, gfx.Rect{X: bounds.X + 8, Y: bounds.Y, Width: navigationMenuMax(0, bounds.Width-16), Height: bounds.Height}, gfx.TextStyle{Size: content.theme.Metrics.FontSizeM, Weight: gfx.WeightMedium, Color: foreground})
	} else {
		content.canvas.Label(link.Label, gfx.Rect{X: bounds.X + 8, Y: bounds.Y + 7, Width: navigationMenuMax(0, bounds.Width-16), Height: 18}, gfx.TextStyle{Size: content.theme.Metrics.FontSizeM, Weight: gfx.WeightMedium, Color: foreground})
		content.canvas.Label(link.Description, gfx.Rect{X: bounds.X + 8, Y: bounds.Y + 26, Width: navigationMenuMax(0, bounds.Width-16), Height: 18}, gfx.TextStyle{Size: content.theme.Metrics.FontSizeS, Weight: gfx.WeightRegular, Color: description})
	}
	if !link.ID.IsZero() {
		content.canvas.HitWithOptions(link.ID, bounds, link.Label, gfx.HitOptions{Role: gfx.RoleLink, Value: link.Value, Hint: link.Hint, Disabled: link.Disabled, Selected: link.Active})
	}
	content.pop()
	return bounds
}

// AddSeparator paints a content divider.
func (content *NavigationMenuContent) AddSeparator() gfx.Rect {
	if !content.visible || content.y+9 > content.bounds.Y+content.bounds.Height-8 {
		return gfx.Rect{}
	}
	bounds := gfx.Rect{X: content.bounds.X + 8, Y: content.y, Width: navigationMenuMax(0, content.bounds.Width-16), Height: 9}
	content.y += 9
	content.push()
	content.canvas.Fill(gfx.Rect{X: bounds.X, Y: bounds.Y + 4, Width: bounds.Width, Height: 1}, content.theme.Border)
	content.pop()
	return bounds
}

func (content NavigationMenuContent) push() {
	content.canvas.PushLayer(content.options.Layer)
	if !content.options.Scope.IsZero() {
		content.canvas.PushFocusScope(content.options.Scope)
	}
}
func (content NavigationMenuContent) pop() {
	if !content.options.Scope.IsZero() {
		content.canvas.PopFocusScope()
	}
	content.canvas.PopLayer()
}
func navigationMenuMax(a, b float32) float32 {
	if a > b {
		return a
	}
	return b
}

// NavigationMenuHover advances the navigation menu's hover switch: while a
// panel is open, resting on another trigger opens it after delay. It returns
// the open trigger and the wait before asking again.
func NavigationMenuHover(open, trigger gfx.ID, now time.Time, intent *HoverIntent, delay time.Duration) (gfx.ID, time.Duration) {
	if open.IsZero() || trigger.IsZero() || trigger == open {
		intent.Reset()
		return open, 0
	}
	id, wait := intent.Rest(trigger, now, delay)
	if id.IsZero() {
		return open, wait
	}
	return id, 0
}
