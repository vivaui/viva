package {{PACKAGE}}

import (
	"github.com/vivaui/viva/gfx"
)

// RadioOption is one stable choice in a RadioGroup.
type RadioOption struct {
	ID       gfx.ID
	Label    string
	Disabled bool
	Invalid  bool
}

// RadioGroupOrientation selects row or column layout.
type RadioGroupOrientation uint8

const (
	RadioGroupVertical RadioGroupOrientation = iota
	RadioGroupHorizontal
)

// RadioGroup paints a single-selection set. Selection remains application
// state; each enabled option is an independently focusable stable hit.
type RadioGroup struct {
	canvas      *gfx.Canvas
	theme       gfx.Theme
	bounds      gfx.Rect
	options     []RadioOption
	selected    gfx.ID
	orientation RadioGroupOrientation
	disabled    bool
	// tabStop is the one option Tab reaches; the arrow helper can focus the
	// others, which are SkipTab.
	tabStop gfx.ID
}

// RadioGroupResult is what painting a radio group reports back.
type RadioGroupResult struct {
	canvas  *gfx.Canvas
	options []RadioOption
}

// Clicked returns the option whose click this paint is delivering, and false in
// every other paint (see gfx.Canvas.Clicked), so the code that declares the group
// is the code that moves the selection:
//
//	if id, ok := components.BeginRadioGroup(canvas, theme, bounds, options, selected).Paint().Clicked(); ok {
//		selected = id
//	}
func (result RadioGroupResult) Clicked() (gfx.ID, bool) {
	if result.canvas == nil {
		return gfx.ID{}, false
	}
	for _, option := range result.options {
		if result.canvas.Clicked(option.ID) {
			return option.ID, true
		}
	}
	return gfx.ID{}, false
}

// BeginRadioGroup starts a vertical radio group.
func BeginRadioGroup(canvas *gfx.Canvas, theme gfx.Theme, bounds gfx.Rect, options []RadioOption, selected gfx.ID) RadioGroup {
	group := RadioGroup{canvas: canvas, theme: theme, bounds: bounds, options: options, selected: selected}
	group.tabStop = group.rovingTabStop()
	return group
}

// rovingTabStop keeps the Tab stop on the option the user last focused, else on
// the selection, else on the first enabled option.
func (group RadioGroup) rovingTabStop() gfx.ID {
	enabled := func(option RadioOption) bool {
		return !group.disabled && !option.Disabled && !option.ID.IsZero()
	}
	focus := gfx.ID{}
	if group.canvas != nil {
		focus = group.canvas.FocusVisible()
	}
	for _, option := range group.options {
		if option.ID == focus && enabled(option) {
			return focus
		}
	}
	for _, option := range group.options {
		if option.ID == group.selected && enabled(option) {
			return group.selected
		}
	}
	for _, option := range group.options {
		if enabled(option) {
			return option.ID
		}
	}
	return gfx.ID{}
}

// Horizontal lays choices out in equal-width columns.
func (group RadioGroup) Horizontal() RadioGroup {
	group.orientation = RadioGroupHorizontal
	return group
}

// Disabled renders the complete group as inert.
func (group RadioGroup) Disabled() RadioGroup {
	group.disabled = true
	return group
}

// RadioGroupKeyDown applies radio-group keyboard navigation to current.
// Arrow keys wrap through enabled non-zero options. Physical Left and Right
// reverse in RTL. Base UI leaves Home and End to the page, so they are not
// owned here. The returned bool reports whether key is owned by the pattern.
func RadioGroupKeyDown(current gfx.ID, options []RadioOption, direction Direction, key gfx.Key) (gfx.ID, bool) {
	step := 0
	switch key {
	case gfx.KeyLeft:
		step = -1
		if direction.IsRTL() {
			step = 1
		}
	case gfx.KeyRight:
		step = 1
		if direction.IsRTL() {
			step = -1
		}
	case gfx.KeyUp:
		step = -1
	case gfx.KeyDown:
		step = 1
	default:
		return current, false
	}
	return nextEnabledRadioOption(current, options, step), true
}

func firstEnabledRadioOption(current gfx.ID, options []RadioOption, step int) gfx.ID {
	for index := 0; index < len(options); index++ {
		candidate := index
		if step < 0 {
			candidate = len(options) - 1 - index
		}
		option := options[candidate]
		if !option.ID.IsZero() && !option.Disabled {
			return option.ID
		}
	}
	return current
}

func nextEnabledRadioOption(current gfx.ID, options []RadioOption, step int) gfx.ID {
	if len(options) == 0 {
		return current
	}
	currentIndex := -1
	for index := range options {
		if options[index].ID == current {
			currentIndex = index
			break
		}
	}
	if currentIndex < 0 {
		return firstEnabledRadioOption(current, options, step)
	}
	for offset := 1; offset <= len(options); offset++ {
		index := (currentIndex + step*offset) % len(options)
		if index < 0 {
			index += len(options)
		}
		option := options[index]
		if !option.ID.IsZero() && !option.Disabled {
			return option.ID
		}
	}
	return current
}

// Paint draws choices and declares their hit regions. The circle sits on the
// left edge of its row and the label one item gap after it.
func (group RadioGroup) Paint() RadioGroupResult {
	result := RadioGroupResult{canvas: group.canvas, options: group.options}
	if group.canvas == nil || len(group.options) == 0 || group.bounds.Width <= 0 {
		return result
	}
	metrics := group.theme.Metrics
	rowHeight := metrics.ControlHeight
	gap := float32(12)
	if group.orientation == RadioGroupVertical && group.bounds.Height > 0 {
		available := group.bounds.Height - gap*float32(len(group.options)-1)
		if candidate := available / float32(len(group.options)); candidate > 0 {
			rowHeight = candidate
		}
	}
	if group.orientation == RadioGroupHorizontal && group.bounds.Height > 0 {
		rowHeight = group.bounds.Height
	}

	for index, option := range group.options {
		row := group.radioBounds(index, rowHeight, gap)
		enabled := !group.disabled && !option.Disabled && !option.ID.IsZero()
		focused := enabled && group.canvas.FocusVisible() == option.ID
		selected := !option.ID.IsZero() && option.ID == group.selected

		diameter := float32(16)
		circle := gfx.Rect{X: row.X, Y: row.Y + (row.Height-diameter)/2, Width: diameter, Height: diameter}
		border := group.theme.Input
		fill := group.theme.Background
		fill.Alpha = 0
		if isDark(&group.theme) {
			fill = scaledAlpha(group.theme.Input, 0.30)
		}
		if selected {
			border = group.theme.Primary
			fill = group.theme.Primary
		}
		switch {
		case option.Invalid && enabled:
			// The ring is for a control the user can act on; a selected invalid
			// option keeps its primary border and still shows it.
			group.canvas.Rounded(expand(circle, 3), diameter/2+3, invalidRing(&group.theme))
			if !selected {
				border = invalidBorder(&group.theme)
			}
		case focused:
			group.canvas.Rounded(expand(circle, 3), diameter/2+3, scaledAlpha(group.theme.Ring, 0.50))
			if !selected {
				border = group.theme.Ring
			}
		case option.Invalid && !selected:
			border = invalidBorder(&group.theme)
		}
		if group.disabled || option.Disabled {
			border.Alpha *= 0.50
			fill.Alpha *= 0.50
		}
		paintBordered(group.canvas, group.theme, circle, diameter/2, border, fill)
		indicatorSize := float32(8)
		indicator := group.theme.PrimaryForeground
		if !selected {
			indicator.Alpha = 0
		}
		if (group.disabled || option.Disabled) && selected {
			indicator.Alpha *= 0.50
		}
		group.canvas.Rounded(gfx.Rect{
			X: circle.X + (diameter-indicatorSize)/2, Y: circle.Y + (diameter-indicatorSize)/2,
			Width: indicatorSize, Height: indicatorSize,
		}, indicatorSize/2, indicator)

		foreground := group.theme.Foreground
		if group.disabled || option.Disabled {
			foreground.Alpha *= 0.50
		}
		group.canvas.Label(option.Label, gfx.Rect{
			X: circle.X + circle.Width + metrics.GapItem, Y: row.Y,
			Width:  row.X + row.Width - circle.X - circle.Width - metrics.GapItem,
			Height: row.Height,
		}, gfx.TextStyle{Size: metrics.FontSizeM, Weight: gfx.WeightMedium, Color: foreground})
		if !option.ID.IsZero() {
			group.canvas.HitWithOptions(option.ID, row, option.Label, gfx.HitOptions{
				Role: gfx.RoleRadio, Value: option.Label,
				Disabled: group.disabled || option.Disabled,
				Checked:  selected, Selected: selected,
				SkipTab: option.ID != group.tabStop,
			})
		}
	}
	return result
}

func (group RadioGroup) radioBounds(index int, rowHeight, gap float32) gfx.Rect {
	if group.orientation == RadioGroupHorizontal {
		count := float32(len(group.options))
		left := group.bounds.X + group.bounds.Width*float32(index)/count
		right := group.bounds.X + group.bounds.Width*float32(index+1)/count
		return gfx.Rect{X: left, Y: group.bounds.Y, Width: right - left, Height: rowHeight}
	}
	return gfx.Rect{
		X: group.bounds.X, Y: group.bounds.Y + float32(index)*(rowHeight+gap),
		Width: group.bounds.Width, Height: rowHeight,
	}
}
