package main

import (
	"fmt"
	"log"

	"github.com/vivaui/viva/app"
	"github.com/vivaui/viva/flex"
)

// model is everything the window shows. page only reads it; a click handled where
// its button is declared is the only place that changes it.
type model struct {
	count int
}

// page declares the window. It runs again on every repaint, from the model.
func (m *model) page(s flex.Screen) {
	s.Column().Fill().Pad(24).Gap(12).Children(func() {
		s.Text("{{NAME}}").FontSize(20).Bold()
		s.Paragraph("Edit main.go and save: viva dev reloads the window and keeps this count.", 3).Muted()
		s.Spacer()
		s.Text(fmt.Sprintf("Count: %d", m.count)).FontSize(28).Bold()
		s.Row().Gap(8).Children(func() {
			if s.Button("Add one").Clicked() {
				m.count++
			}
			if s.Button("Reset").Outline().Clicked() {
				m.count = 0
			}
		})
	})
}

func config() app.Config {
	return app.Config{Title: "{{NAME}}", Width: 420, Height: 260}
}

func main() {
	m := &model{}
	if err := flex.Run(config(), m.page); err != nil {
		log.Fatal(err)
	}
}
