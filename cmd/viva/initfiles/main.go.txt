package main

import (
	"fmt"
	"log"

	"github.com/vivaui/viva/app"
	"github.com/vivaui/viva/ui"
)

// model is everything the window shows. page only reads it; a click handled where
// its button is declared is the only place that changes it.
type model struct {
	count int
}

// page declares the window. It runs again on every repaint, from the model.
func (m *model) page(c ui.C) {
	ui.Column(c).Fill().Pad(24).Gap(12).Children(func() {
		ui.Text(c, "{{NAME}}").FontSize(20).Bold()
		ui.Paragraph(c, "Edit main.go and save: viva dev reloads the window and keeps this count.", 3).Muted()
		ui.Spacer(c)
		ui.Text(c, fmt.Sprintf("Count: %d", m.count)).FontSize(28).Bold()
		ui.Row(c).Gap(8).Children(func() {
			if ui.Button(c, "Add one").Clicked() {
				m.count++
			}
			if ui.Button(c, "Reset").Outline().Clicked() {
				m.count = 0
			}
		})
	})
}

func config() app.Config {
	return app.Config{Title: "{{NAME}}", Width: 420, Height: 260}
}

func main() {
	m := &model{}
	if err := ui.Run(config(), m.page); err != nil {
		log.Fatal(err)
	}
}
