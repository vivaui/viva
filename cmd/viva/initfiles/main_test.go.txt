package main

import (
	"fmt"
	"strings"
	"testing"

	"github.com/vivaui/viva/app"
	"github.com/vivaui/viva/flex"
)

// open runs the page with no window: the real painter, input and focus rules.
func open(t *testing.T, m *model) *app.Driver {
	t.Helper()
	screen := flex.NewScreen(flex.ScreenConfig{Body: m.page})
	driver, err := app.NewDriver(screen.Config(config()), nil)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(driver.Close)
	return driver
}

// click presses the middle of the control with the given name, as a user would.
func click(t *testing.T, driver *app.Driver, name string) {
	t.Helper()
	for _, hit := range driver.Snapshot().Hits {
		if hit.Label == name && !hit.IsLabel {
			b := hit.Bounds
			if err := driver.Click(b.X+b.Width/2, b.Y+b.Height/2); err != nil {
				t.Fatal(err)
			}
			return
		}
	}
	t.Fatalf("no control named %q on screen", name)
}

func shows(driver *app.Driver, want string) bool {
	for _, label := range driver.Snapshot().Labels {
		if strings.Contains(label.Value, want) {
			return true
		}
	}
	return false
}

func TestButtonsChangeTheCount(t *testing.T) {
	m := &model{}
	driver := open(t, m)
	for want := 1; want <= 3; want++ {
		click(t, driver, "Add one")
		if !shows(driver, fmt.Sprintf("Count: %d", want)) {
			t.Fatalf("after %d clicks the window does not show the count", want)
		}
	}
	click(t, driver, "Reset")
	if !shows(driver, "Count: 0") {
		t.Fatal("Reset did not zero the count")
	}
}
